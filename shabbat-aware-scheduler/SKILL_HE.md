### שלב 2: קבלת זמני הלכה ונתוני חגים

תשתמשו ב-HebCal API כדי לשלוף זמני שבת ונתוני חגים.
תסתכלו על `scripts/check_shabbat.py` לכלי מוכן לשימוש.

**שאילתת HebCal API לזמני שבת:**

עדיף להעביר `geonameid` ולא קואורדינטות גולמיות. כשמעבירים geonameid, Hebcal מיישם לבד את
מנהג הדלקת הנרות של אותה עיר, ואין ערך `b=` שאפשר לטעות בו. קואורדינטות עם `b=` ידני שמורות
למקום שאין לו geonameid.

```python
import requests
from datetime import datetime, timedelta

CITY_GEONAMEID = {
    "jerusalem": 281184, "tel_aviv": 293397, "haifa": 294801,
    "beersheba": 295530, "tiberias": 293322, "safed": 293100,
}

def get_shabbat_times(date=None, geonameid=281184):
    """זמני הדלקת נרות והבדלה לשבת של השבוע שבו נמצא `date`."""
    if date is None:
        date = datetime.now()

    # מעגנים על יום שישי של שבוע השבת הנוכחי. שבת שייכת לשבת שנכנסה בערב הקודם,
    # ולכן מסתכלים יום אחד אחורה. חישוב (4 - weekday) % 7 ללא תנאי מחזיר לאורך
    # כל יום שבת את השבת של השבוע הבא, כלומר טעות של 8 ימים בדיוק ברגע שבו
    # מישהו שואל "מתי שבת יוצאת?".
    if date.weekday() == 5:
        friday = date - timedelta(days=1)
    else:
        friday = date + timedelta(days=(4 - date.weekday()) % 7)

    response = requests.get("https://www.hebcal.com/shabbat", params={
        "cfg": "json",
        "gy": friday.year, "gm": friday.month, "gd": friday.day,
        "geonameid": geonameid,
        "M": "on",    # הבדלה לפי צאת הכוכבים (השמש 8.5 מעלות מתחת לאופק)
        # חלופה: m=50 או m=72 למספר קבוע של דקות אחרי השקיעה
    })

    data = response.json()
    times = {}
    for item in data.get("items", []):
        if item["category"] == "candles":
            times["candle_lighting"] = item["date"]
        elif item["category"] == "havdalah":
            times["havdalah"] = item["date"]

    return times
```

ברירת המחדל של `b=` ב-Hebcal היא 18 דקות לפני השקיעה, 40 לירושלים ו-30 לחיפה ולזכרון יעקב,
ואותה הסטה חלה גם על ערב חג (ראש השנה, יום כיפור, סוכות, שמיני עצרת, שמחת תורה, פסח, שבועות)
ולא רק על ערב שבת.

**קבלת כל החגים הישראליים לשנה:**
```python
def get_holidays(year):
    """Get all Israeli holidays for a given year."""
    response = requests.get("https://www.hebcal.com/hebcal", params={
        "v": 1,
        "cfg": "json",
        "year": year,
        "month": "x",  # All months
        "maj": "on",   # Major holidays
        "min": "on",   # Minor holidays
        "mod": "on",   # Modern holidays
        "mf": "on",    # צומות קלים. בלי הדגל הזה, צום גדליה, עשרה בטבת,
                       # תענית אסתר, תענית בכורות וצום תמוז לא מופיעים
                       # בתשובה בכלל.
        "i": "on",     # Israeli holidays (1-day yom tov)
        "nx": "off",
        "ss": "off"
    })

    data = response.json()
    holidays = []
    for item in data.get("items", []):
        if item["category"] in ["holiday", "roshchodesh"]:
            holidays.append({
                "title": item["title"],
                "date": item["date"],
                "category": item.get("subcat", item["category"]),
                "yomtov": item.get("yomtov", False),
                "memo": item.get("memo", "")
            })

    return holidays
```

**פונקציית תזמון מרכזית:**
```python
from datetime import datetime, timedelta, time
import pytz

IL_TZ = pytz.timezone("Asia/Jerusalem")

BUSINESS_HOURS = {
    6: (time(8, 0), time(18, 0)),  # Sunday
    0: (time(8, 0), time(18, 0)),  # Monday
    1: (time(8, 0), time(18, 0)),  # Tuesday
    2: (time(8, 0), time(18, 0)),  # Wednesday
    3: (time(8, 0), time(18, 0)),  # Thursday
    4: (time(8, 0), time(13, 0)),  # Friday (half day)
    5: None,                        # Saturday (Shabbat)
}

# ימים שבהם המדינה לא עובדת אבל Hebcal מסמן אותם yomtov=false.
# סינון לפי yomtov בלבד הופך כל אחד מהם ליום עסקים רגיל.
CLOSED_OBSERVANCES = {
    "Yom HaAtzma'ut": "closed",   "Yom HaZikaron": "closed",
    "Tish'a B'Av": "reduced",     "Erev Yom Kippur": "reduced",
    "Erev Rosh Hashana": "reduced", "Erev Pesach": "reduced",
    "Erev Sukkot": "reduced",     "Erev Shavuot": "reduced",
}
JERUSALEM_ONLY = {"Shushan Purim"}   # ערים מוקפות חומה בלבד

def _norm(title):
    # Hebcal מחזיר גרש טיפוגרפי, ולכן "Yom HaAtzma\u2019ut" לעולם לא יתאים לטבלה
    # שהמפתחות בה בגרש רגיל. התו הבודד הזה הוא כל הבאג.
    return title.replace("\u2019", "'").strip()

def classify_day(date, holidays_cache=None, jerusalem=False):
    """מחזיר (status, reason).

    closed         שבת, יום טוב, או יום השבתה ארצי
    reduced        בוקר בלבד (ערב חג, יום צום)
    evening_closed יום עבודה מלא שהערב שלו מוגבל: ערב יום הזיכרון, ערב יום
                   השואה, ערב תשעה באב. נפרד מ-reduced בכוונה, אחרת החיתוך
                   של "בוקר בלבד" היה חוסם צהריים של יום שני רגיל.
    open           יום עסקים רגיל
    """
    if date.weekday() == 5:
        return "closed", "Shabbat"
    if not holidays_cache:
        return "open", "Business day (no holiday data loaded)"

    date_str = date.strftime("%Y-%m-%d")
    status, reason = "open", "Business day"
    for h in holidays_cache:
        if not h["date"].startswith(date_str):
            continue
        title = _norm(h["title"])
        if h.get("yomtov", False):
            return "closed", f"Yom Tov: {title}"
        if title in JERUSALEM_ONLY:
            if jerusalem:
                return "closed", f"{title} (Jerusalem and walled cities)"
            continue
        level = CLOSED_OBSERVANCES.get(title)
        if level == "closed":
            return "closed", title
        if level == "reduced" and status == "open":
            status, reason = "reduced", f"{title} (morning only)"
    return status, reason
```

חול המועד מסווג `open`, כי מבחינת החוק בישראל זה יום עבודה, גם אם האיוש דליל. בכל מקרה
מסופק אחר, מכריעים לכיוון `closed`. הצעת משבצת שמתברר שהיא סגורה עולה פגישה שנקבעת מחדש;
פספוס של סגירה עולה את הפגישה עצמה.

### שלב 4: משימות cron מודעות חגים

**בדיקת תאריך היא לא בדיקת זמן.** `classify_day()` עונה על שאלה לגבי תאריך בלוח. שבת ויום טוב
נכנסים בהדלקת הנרות בערב הקודם, ולכן יום שישי הוא תאריך עסקים בזמן ששישי ב-19:00 הוא כבר שבת.
כל דבר שמופעל ברגע מסוים (cron, שליחת התראה, פריסה) חייב לבדוק גם את החלון החי, אחרת הוא ירוץ
בשבת כל שבוע מחדש ויכתוב ביומן "Business day".

```python
def is_shabbat_now(now, times):
    """times = get_shabbat_times(now). אם ה-API לא זמין, נועלים ולא מריצים."""
    start = datetime.fromisoformat(times["candle_lighting"])
    end = datetime.fromisoformat(times["havdalah"])
    return start <= now <= end
```

```python
def should_run_today(holidays_cache=None, skip_friday=False, skip_erev_chag=False,
                     now=None, jerusalem=False, time_aware=True, rest_kwargs=None):
    """Determine if a scheduled job should run RIGHT NOW.

    rest_kwargs עובר לבדיקת החלון החי (geonameid, או עיר וקואורדינטות).
    בלעדיו החלון מחושב לפי ירושלים, כלומר 22 דקות מוקדם מדי למי שבתל אביב.
    """
    today = datetime.now(IL_TZ).date()

    # לעולם לא בשבת - החלון החי, לא רק היום בשבוע
    if is_shabbat_now(datetime.now(IL_TZ), get_shabbat_times()):
        return False, "Shabbat in progress"
    if today.weekday() == 5:
        return False, "Shabbat"

    # Check holidays
    if holidays_cache:
        date_str = today.strftime("%Y-%m-%d")
        status, reason = classify_day(today, holidays_cache)
        if status == "closed":
            return False, reason

        # Check if tomorrow is Yom Tov (today is Erev Chag)
        if skip_erev_chag:
            tomorrow = today + timedelta(days=1)
            tomorrow_str = tomorrow.strftime("%Y-%m-%d")
            for h in holidays_cache:
                if h["date"].startswith(tomorrow_str) and h["yomtov"]:
                    return False, f"Erev Chag: {h['title']} tomorrow"

    if skip_friday and today.weekday() == 4:
        return False, "Friday (half day)"

    return True, "Business day"
```

## דוגמאות

### דוגמה 1: קביעת פגישה
המשתמש אומר: "קבע פגישת צוות לשבוע הבא"
תוצאה: בודקים שעות עבודה ישראליות (א'-ה'), מאמתים שאין חגים, מציעים משבצות זמינות. נמנעים מיום שישי אלא אם בבוקר ומאשרים שזה לא ערב חג.

### דוגמה 2: חלון פריסה
המשתמש אומר: "מתי הזמן הבטוח ביותר לפרוס השבוע?"
תוצאה: מוצאים משבצת בשלישי או ברביעי (אמצע שבוע, הכי רחוק משבת), בשעות העבודה, לא לפני חג. ממליצים על פריסה בבוקר לזמן rollback מקסימלי לפני שבת.

### דוגמה 3: Cron מודע חגים
המשתמש אומר: "הגדר דו"ח יומי שמדלג על שבתות וחגים"
תוצאה: מגדירים cron עם בדיקת should_run_today(), מטמון חגים טעון מראש לשנה, עם לוגים לימים שדולגו.

## ספריות לעבודה ללא רשת

אם אי אפשר להגיע ל-Hebcal API בזמן ריצה (CI, סביבה מנותקת, הגבלת קצב של 90 בקשות ל-10 שניות), עדיף להשתמש בספרייה מקומית ולוותר על קריאת ה-HTTP:

| ספרייה | שפה | הערות |
|---------|------|-------|
| `@hebcal/core` | JavaScript / TypeScript | בתחזוקה פעילה (v6.x נכון למאי 2026). JS טהור, בלי רשת. התקנה: `npm i @hebcal/core` |
| `pyluach` | Python | חשבון לוח עברי והמרת תאריכים עברי-גרגוריאני. יציבה אבל בפעילות נמוכה (v2.3.0); מתאימה להמרת תאריכים אבל בלי זמני הדלקת נרות מובנים. כדאי לשלב עם ספריית זמנים או עם זמנים מחושבים מראש במטמון |
| `hebcal-go` | Go | מתוחזקת על ידי צוות Hebcal |

החבילה המיושנת `hebcal-js` (בשם `hebcal` ב-NPM) היא הקודמת של `@hebcal/core`; אין להתחיל איתה עבודה חדשה.

## מלכודות נפוצות
- הדלקת נרות שונה לפי עיר בישראל. ירושלים 40 דקות לפני שקיעה, חיפה וזכרון יעקב 30 דקות, ובשאר הערים 18 דקות. שימוש בערך אחיד לכל ישראל יחטיא את זמן הסיום של יום שישי ב-10 עד 22 דקות. מעבירים ל-API את הערך לפי העיר.
- לחגים ישראליים יש מגבלות עבודה שונות משבת. רוב החגים הם יום אחד בישראל אבל יומיים בחו"ל (ראש השנה הוא יומיים בשניהם). שימוש בלוח חגים של חו"ל יחסום ימי עבודה שהם בעצם חול המועד בישראל.
- בלוח העברי יש שנים מעוברות עם חודש נוסף (אדר ב'), שמתרחשות 7 פעמים במחזור של 19 שנה. סוכנים עלולים לחשב תאריכים לפי הלוח הגרגוריאני ולפספס את החודש הזה.
- שעות העבודה בישראל הן ראשון עד חמישי, ושישי הוא חצי יום (עד אחרי הצהריים מוקדם). שבת היא יום המנוחה השבועי, לא ראשון. סוכנים עלולים לתזמן פגישות בשישי אחרי הצהריים או דדליינים לבוקר יום שני.
- יום העצמאות ויום הזיכרון יכולים להידחות (נדחה) כשהתאריך הטבעי מתנגש עם שבת. בשנת 2026 אין דחייה (ה' באייר תשפ"ו חל ב-21 עד 22 באפריל). כדאי לקרוא את התאריך חזרה מ-Hebcal ולא לחשב ה' באייר ידנית. Hebcal מיישם את הדחייה בעצמו; `i=on` הוא דגל נפרד שבוחר יום טוב ישראלי של יום אחד, והוא לא זה שמבצע את ההזזה.
- ברירת המחדל של Hebcal להבדלה היא צאת הכוכבים (השמש 8.5 מעלות מתחת לאופק). מדידה מול endpoint הזמנים של Hebcal עצמו לירושלים נותנת בערך 37 דקות אחרי השקיעה בסוף הקיץ ו-39 עד 40 דקות באמצע החורף, ולא 50 דקות כפי שמצטטים לעיתים קרובות. שומרי מנהג רבנו תם משתמשים ב-72 דקות. בוחרים את ערך `m=` המתאים אם הקהל שלכם לא ברירת המחדל.
- Hebcal מסמן `yomtov: true` רק על יום טוב. יום העצמאות, יום הזיכרון, ערב יום כיפור, תשעה באב וכל ערב חג חוזרים עם `yomtov: false`, ולכן סינון שנכתב כ-`if h["yomtov"]` מתייחס ליום העצמאות כאל יום רביעי רגיל. צריך להתאים גם לפי הכותרת, ולשים לב ש-Hebcal כותב את הכותרות האלה עם גרש טיפוגרפי (`Yom HaAtzma\u2019ut`), כך שחיפוש בטבלה עם גרש רגיל פשוט לעולם לא נדלק.
- צומות קלים לא נראים בכלל בלי הדגל `mf=on`. הדגלים `maj` / `min` / `mod` לא מחזירים את צום גדליה, עשרה בטבת, תענית אסתר, תענית בכורות או צום תמוז. צום גדליה נופל למחרת ראש השנה, בדיוק כשמישהו קובע את פגישת ההשלמות.
- כשנתוני הלוח חסרים, נועלים ולא פותחים. ל-Hebcal יש הגבלת קצב של 90 בקשות ל-10 שניות, ומתזמן שעונה "יום עסקים" על 429 יקבע פגישה ביום כיפור. מטמון חגים ריק הוא "לא ידוע ולכן סגור", אף פעם לא "פתוח".
- סטטוס ערב חג צריך להיגזר ("ליום הבא יש `yomtov: true`") ולא להיות רשימה קשיחה. רשימה קבועה מפספסת מקרים כמו הושענא רבה, שהוא ערב שמיני עצרת בכל שנה.
- שושן פורים (ט"ו באדר) הוא חג מלא בירושלים ובערים מוקפות חומה, ויום עבודה רגיל בכל מקום אחר. מתזמן ארצי לא אמור לחסום אותו, ומתזמן ירושלמי לא אמור להתעלם ממנו. זו התצפית הישראלית היחידה שהסטטוס שלה תלוי בעיר ולא בתאריך.
- כלל שעון הקיץ בישראל הוא ייחודי: שעון קיץ מתחיל ביום שישי שלפני יום ראשון האחרון של מרץ ומסתיים ביום ראשון האחרון של אוקטובר, כלומר לא כמו ארצות הברית ולא כמו אירופה. בשנת 2026 מדובר ב-27 במרץ וב-25 באוקטובר. שני המעברים מזיזים את הדלקת הנרות של יום שישי בשעה שלמה בשעון הקיר, והמעבר של אוקטובר הוא המלכודת החדה יותר, כי פתאום שבת נכנסת שעה מוקדם יותר. עדיף להישען על מסד הנתונים `Asia/Jerusalem` ולא על היסט קשיח.
- יום כיפור נחשב כמו שבת לצרכי תזמון (השבתה מלאה כולל עסקים, תחבורה ושידור בישראל). לא לפרוס ולא לתזמן כלום בחלון של 25 השעות האלה.
- בין המצרים (י"ז בתמוז עד ט' באב) היא תקופת אבל; חתונות, הופעות ואירועי חברה בדרך כלל נדחים. תשעת הימים (א' עד ט' באב) מחמירה יותר. זה לא יום טוב אבל זה חלון של "בלי חגיגות".

## פתרון בעיות

### שגיאה: "פגישה נקבעה בזמן שבת"
סיבה: אי-התאמה של אזור זמן - השרת ב-UTC, זמני שבת בשעון המקומי
פתרון: תמיד להמיר לאזור הזמן Asia/Jerusalem לפני הבדיקה. זמני שבת משתנים לפי העונה והמיקום.

### שגיאה: "חג לא זוהה"
סיבה: שימוש בלוח גרגוריאני בלבד בלי מיפוי של תאריכים עבריים
פתרון: תשתמשו ב-HebCal API שמטפל בהמרה עברי-גרגוריאני. תשמרו במטמון נתוני חגים שנתיים ותרעננו בראש השנה.

### שגיאה: "פגישת יום שישי מאוחרת מדי"
סיבה: זמן סיום קבוע של 17:00 ביום שישי בלי להתחשב בעונה
פתרון: בחורף, שבת יכולה להתחיל כבר ב-16:00. תמיד תבדקו את זמן הדלקת הנרות בפועל ליום שישי הספציפי.

### שגיאה: "זמן הדלקת נרות שגוי בירושלים"
סיבה: שולחים ל-Hebcal `b=18` (ברירת המחדל) עם קואורדינטות של ירושלים, במקום מנהג ירושלים של 40 דקות.
פתרון: שולחים `b=40` כשהמשתמש בירושלים, `b=30` לחיפה וזכרון יעקב, ו-`b=18` בכל מקום אחר. Hebcal עצמו עובד באותה מוסכמה.

### שגיאה: "זמן הבדלה חורג ב-8 עד 30 דקות"
סיבה: ערבוב בין `M=on` (צאת הכוכבים, השמש 8.5 מעלות מתחת לאופק, שנמדד ב-37 עד 40 דקות אחרי השקיעה בירושלים לאורך השנה) לבין הנחה קשיחה של "42 דקות" או "72 דקות".
פתרון: בוחרים שיטה אחת ובאופן מפורש. `M=on` הוא צאת הכוכבים של Hebcal (השמש 8.5 מעלות מתחת לאופק, שלושה כוכבים קטנים נראים). `m=N` הוא לעומת זאת מספר דקות קבוע אחרי השקיעה, ולכן `m=50` ו-`m=72` הם ספירת דקות ולא ספירת כוכבים. מתעדים בקוד באיזו שיטה משתמש המתזמן כדי שסוכן אחר לא יוסיף עוד הסטה.

### שגיאה: "פגישה נקבעה ביום העצמאות / בתשעה באב / בערב חג"
סיבה: סינון של פיד Hebcal לפי `yomtov == true` בלבד. נבדק מול הפיד הישראלי לשנת 2026: יום הזיכרון (21.4), יום העצמאות (22.4), ערב יום כיפור (20.9) ותשעה באב (23.7) כולם מחזירים `yomtov: false`.
פתרון: לסווג לפי הכותרת בנוסף לדגל, ולנרמל קודם את הגרש הטיפוגרפי. ראו `classify_day()` ב-`scripts/check_shabbat.py`.

### שגיאה: "משימת cron רצה בשישי ב-19:30, בתוך שבת"
סיבה: הבדיקה התייחסה ליום בשבוע, ויום שישי הוא תאריך עסקים עד הדלקת הנרות.
פתרון: להשוות את הרגע הנוכחי מול החלון שבין הדלקת נרות להבדלה שמחזיר Hebcal (`is_shabbat_now()`), ולנעול ולא להריץ בסוף השבוע אם ה-API לא זמין.
